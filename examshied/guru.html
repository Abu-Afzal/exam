Belum ada soal di bank

'; return; } container.innerHTML = soalList.map((s, i) => `
Soal ${i+1} (${s.jenis})

${s.teks_soal}

`).join(''); } catch (error) { console.error('Error refresh bank soal:', error); } } window.hapusSoal = async (id) => { if (!confirm('Hapus soal ini?')) return; try { const { error } = await supabaseClient.from('bank_soal').delete().eq('id', id); if (error) throw error; alert('✅ Soal dihapus'); await refreshBankSoal(); await refreshStats(); } catch (error) { alert('❌ Gagal hapus: ' + error.message); } }; // Populate mapel dropdown const bsMapel = document.getElementById('bsMapel'); if (bsMapel && currentUser.profile.mapel) { bsMapel.innerHTML = ``; } // Form Input Soal (Batch) let soalBatch = []; function renderSoalBatch() { const container = document.getElementById('soalContainer'); if (!container) return; container.innerHTML = soalBatch.map((s, i) => `
Soal ${i+1}
`).join(''); } const btnTambahSoal = document.getElementById('btnTambahSoal'); if (btnTambahSoal) { btnTambahSoal.addEventListener('click', () => { soalBatch.push({ teks_soal: '', opsi: { A:'', B:'', C:'', D:'' }, jawaban: '' }); renderSoalBatch(); }); } const btnSimpanSemuaSoal = document.getElementById('btnSimpanSemuaSoal'); if (btnSimpanSemuaSoal) { btnSimpanSemuaSoal.addEventListener('click', async () => { const mapel = document.getElementById('bsMapel')?.value || currentUser.profile.mapel; const jenis = document.getElementById('bsJenis')?.value || 'pg'; const poin = parseInt(document.getElementById('bsPoin')?.value) || 10; if (soalBatch.length === 0) { alert('Tambahkan minimal 1 soal!'); return; } if (!mapel) { alert('Pilih mata pelajaran!'); return; } try { let berhasil = 0; for (const s of soalBatch) { if (!s.teks_soal || !s.jawaban) continue; const soal = { guru_id: currentUser.profile.id, mapel: mapel, jenis: jenis, teks_soal: s.teks_soal, opsi: s.opsi, jawaban: s.jawaban, poin: poin }; await DB.saveSoal(soal); berhasil++; } alert(`✅ ${berhasil} soal berhasil disimpan ke bank ${mapel}`); soalBatch = []; renderSoalBatch(); document.getElementById('modalSoal').classList.add('hidden'); await refreshBankSoal(); await refreshStats(); } catch (error) { console.error('Error simpan soal:', error); alert('❌ Gagal menyimpan soal: ' + error.message); } }); } // ============================================ // UJIAN SAYA // ============================================ async function refreshUjianGuru() { try { const ujianList = await DB.getUjian(); const ujianGuru = ujianList ? ujianList.filter(u => u.guru_id === currentUser.profile.id) : []; const tbody = document.querySelector('#tabelUjianGuru tbody'); if (!tbody) return; if (ujianGuru.length === 0) { tbody.innerHTML = 'Belum ada ujian'; return; } tbody.innerHTML = ujianGuru.map(u => ` ${u.nama} ${u.kelas} ${u.token} ● ${u.status || 'aktif'} `).join(''); } catch (error) { console.error('Error refresh ujian guru:', error); } } // ============================================ // HASIL UJIAN SISWA // ============================================ async function refreshHasil() { try { const hasilList = await DB.getHasil(); const ujianList = await DB.getUjian(); const ujianGuruIds = ujianList ? ujianList.filter(u => u.guru_id === currentUser.profile.id).map(u => u.id) : []; const hasilGuru = hasilList ? hasilList.filter(h => ujianGuruIds.includes(h.ujian_id)) : []; const tbody = document.querySelector('#tabelHasil tbody'); if (!tbody) return; if (hasilGuru.length === 0) { tbody.innerHTML = 'Belum ada hasil'; return; } tbody.innerHTML = hasilGuru.map(h => { const ujian = ujianList?.find(u => u.id === h.ujian_id); return ` ${h.siswa || '-'} ${ujian?.nama || '-'} ${h.nilai} ${h.submitted_at ? new Date(h.submitted_at).toLocaleString('id-ID') : '-'} `; }).join(''); } catch (error) { console.error('Error refresh hasil:', error); } }
